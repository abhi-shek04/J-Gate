   437→type I18nContextType = {
   438→  lang: Lang;
   439→  setLang: (l: Lang) => void;
   440→  toggle: () => void;
   441→  t: (key: string) => string;
   442→  /** Inline bilingual helper — for page-specific content */
   443→  tx: (entry: { EN: string; JP: string }) => string;
   444→};
   445→
   446→const I18nContext = createContext<I18nContextType | null>(null);
   447→
   448→export function I18nProvider({ children }: { children: ReactNode }) {
   449→  const [lang, setLang] = useState<Lang>("EN");
   450→
   451→  const toggle = useCallback(() => {
   452→    setLang((l) => (l === "EN" ? "JP" : "EN"));
   453→  }, []);
   454→
   455→  const t = useCallback(
   456→    (key: string) => {
   457→      const entry = translations[key];
   458→      if (!entry) return key;
   459→      return entry[lang];
   460→    },
   461→    [lang]
   462→  );
   463→
   464→  const tx = useCallback(
   465→    (entry: { EN: string; JP: string }) => entry[lang],
   466→    [lang]
   467→  );
   468→
   469→  return (
   470→    <I18nContext.Provider value={{ lang, setLang, toggle, t, tx }}>
   471→      {children}
   472→    </I18nContext.Provider>
   473→  );
   474→}
   475→
   476→export function useI18n() {
   477→  const ctx = useContext(I18nContext);
   478→  if (!ctx) {
   479→    return {
   480→      lang: "EN" as Lang,
   481→      setLang: () => {},
   482→      toggle: () => {},
   483→      t: (key: string) => translations[key]?.EN ?? key,
   484→      tx: (entry: { EN: string; JP: string }) => entry.EN,
   485→    };
   486→  }
   487→  return ctx;
   488→}
