   140→/* Cell renderer — ✓ success green, ✗ crimson, △ saffron, ◎ success green */
   141→function Cell({
   142→  type,
   143→  text,
   144→  jgateHighlight,
   145→}: {
   146→  type: CellType;
   147→  text: string;
   148→  jgateHighlight?: boolean;
   149→}) {
   150→  // If no type → plain text
   151→  if (!type) {
   152→    return (
   153→      <span
   154→        className={`font-inter text-[13px] ${jgateHighlight ? "font-semibold text-ink" : "text-slate"}`}
   155→      >
   156→        {text}
   157→      </span>
   158→    );
   159→  }
   160→  // Symbol + text
   161→  const symbol =
   162→    type === "check" ? (
   163→      <Check className="h-3.5 w-3.5 text-success" strokeWidth={2.5} />
   164→    ) : type === "cross" ? (
   165→      <span className="font-serif-jp text-[14px] font-bold text-crimson leading-none">✗</span>
   166→    ) : type === "good" ? (
   167→      <span className="font-serif-jp text-[14px] font-bold text-success leading-none">◎</span>
   168→    ) : (
   169→      <span className="font-serif-jp text-[14px] font-bold text-saffron leading-none">△</span>
   170→    );
   171→  return (
   172→    <div className="flex items-start gap-2">
   173→      <span className="mt-0.5 shrink-0">{symbol}</span>
   174→      <span
   175→        className={`font-inter text-[13px] leading-relaxed ${
   176→          jgateHighlight ? "font-semibold text-ink" : "text-slate"
   177→        }`}
   178→      >
   179→        {text}
   180→      </span>
   181→    </div>
   182→  );
   183→}
