    40→/* 18 CSS particles drifting upward via jg-drift keyframe */
    41→function Particles() {
    42→  const particles = Array.from({ length: 18 }).map((_, i) => ({
    43→    left: `${(i * 53 + 7) % 100}%`,
    44→    size: 1.2 + ((i * 7) % 4) * 0.6,
    45→    delay: `${(i * 1.4) % 22}s`,
    46→    duration: `${18 + ((i * 5) % 12)}s`,
    47→  }));
    48→  return (
    49→    <div className="absolute inset-0 overflow-hidden" aria-hidden>
    50→      {particles.map((p, i) => (
    51→        <span
    52→          key={i}
    53→          className="absolute bottom-0 rounded-full bg-white"
    54→          style={{
    55→            left: p.left,
    56→            width: p.size,
    57→            height: p.size,
    58→            animation: `jg-drift ${p.duration} linear infinite`,
    59→            animationDelay: p.delay,
    60→          }}
    61→        />
    62→      ))}
    63→    </div>
    64→  );
    65→}

    67→/* Floating glass location badge (hero) */
    68→function GlassBadge({
    69→  emoji,
    70→  primary,
    71→  secondary,
    72→  accent,
    73→}: {
    74→  emoji: string;
    75→  primary: string;
    76→  secondary: string;
    77→  accent: "saffron" | "crimson" | "slate";
    78→}) {
    79→  const accentClass =
    80→    accent === "saffron"
    81→      ? "text-saffron border-saffron/30"
    82→      : accent === "crimson"
    83→        ? "text-crimson border-crimson/30"
    84→        : "text-mist border-white/15";
    85→  return (
    86→    <div className="glass-dark flex items-center gap-2.5 rounded-full px-4 py-2">
    87→      <span className="font-serif-jp text-[18px] leading-none">{emoji}</span>
    88→      <div className="flex flex-col leading-tight">
    89→        <span className={`font-inter text-[12px] font-bold uppercase ${accentClass.split(" ")[0]}`} style={{ letterSpacing: "0.08em" }}>
    90→          {primary}
    91→        </span>
    92→        <span className="font-inter text-[11px] text-mist">{secondary}</span>
    93→      </div>
    94→    </div>
    95→  );
    96→}
