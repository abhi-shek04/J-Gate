   448→type I18nContextType = {
   449→  lang: Lang;
   450→  setLang: (l: Lang) => void;
   451→  toggle: () => void;
   452→  t: (key: string) => string;
   453→  /** Inline bilingual helper — for page-specific content */
   454→  tx: (entry: { EN: string; JP: string }) => string;
   455→};
   456→
   457→const I18nContext = createContext<I18nContextType | null>(null);
   458→
   459→export function I18nProvider({ children }: { children: ReactNode }) {
   460→  const [lang, setLang] = useState<Lang>("EN");
   461→
   462→  const toggle = useCallback(() => {
   463→    setLang((l) => (l === "EN" ? "JP" : "EN"));
   464→  }, []);
   465→
   466→  const t = useCallback(
   467→    (key: string) => {
   468→      const entry = translations[key];
   469→      if (!entry) return key;
   470→      return entry[lang];
   471→    },
   472→    [lang]
   473→  );
   474→
   475→  const tx = useCallback(
   476→    (entry: { EN: string; JP: string }) => entry[lang],
   477→    [lang]
   478→  );
   479→
   480→  return (
   481→    <I18nContext.Provider value={{ lang, setLang, toggle, t, tx }}>
   482→      {children}
   483→    </I18nContext.Provider>
   484→  );
   485→}
   486→
   487→export function useI18n() {
   488→  const ctx = useContext(I18nContext);
   489→  if (!ctx) {
   490→    return {
   491→      lang: "EN" as Lang,
   492→      setLang: () => {},
   493→      toggle: () => {},
   494→      t: (key: string) => translations[key]?.EN ?? key,
   495→      tx: (entry: { EN: string; JP: string }) => entry.EN,
   496→    };
   497→  }
   498→  return ctx;
   499→}
