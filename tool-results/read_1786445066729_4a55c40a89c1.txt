   192→/* ── Cell renderer for the comparison table ── */
   193→function Cell({
   194→  type,
   195→  text,
   196→  highlight = false,
   197→}: {
   198→  type?: CellType;
   199→  text: string;
   200→  highlight?: boolean;
   201→}) {
   202→  // If type is set, render icon + text; otherwise render text only
   203→  if (!type) {
   204→    return (
   205→      <span
   206→        className={`font-inter text-[12px] leading-snug sm:text-[13px] ${
   207→          highlight ? "font-semibold text-ink" : "text-slate"
   208→        }`}
   209→      >
   210→        {text}
   211→      </span>
   212→    );
   213→  }
   214→
   215→  let Icon = CheckCircle2;
   216→  let color = "text-success";
   217→  let bg = "bg-success/15";
   218→  if (type === "check") {
   219→    Icon = Check;
   220→    color = highlight ? "text-crimson" : "text-success";
   221→    bg = highlight ? "bg-crimson/15" : "bg-success/15";
   222→  } else if (type === "cross") {
   223→    Icon = X;
   224→    color = "text-slate";
   225→    bg = "bg-slate/10";
   226→  } else if (type === "good") {
   227→    Icon = CircleDot;
   228→    color = highlight ? "text-crimson" : "text-success";
   229→    bg = highlight ? "bg-crimson/15" : "bg-success/15";
   230→  } else if (type === "warn") {
   231→    Icon = Triangle;
   232→    color = "text-saffron";
   233→    bg = "bg-saffron/15";
   234→  }
   235→
   236→  return (
   237→    <div className="flex items-center gap-2.5">
   238→      <span
   239→        className={`flex h-6 w-6 shrink-0 items-center justify-center rounded-full ${bg} ${color}`}
   240→        aria-hidden="true"
   241→      >
   242→        <Icon className="h-3.5 w-3.5" strokeWidth={2.5} />
   243→      </span>
   244→      <span
   245→        className={`font-inter text-[12px] leading-snug sm:text-[13px] ${
   246→          highlight ? "font-semibold text-ink" : "text-slate"
   247→        }`}
   248→      >
   249→        {text}
   250→      </span>
   251→    </div>
   252→  );
   253→}
