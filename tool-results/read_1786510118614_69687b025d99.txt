   141→/* Cell renderer — ✓ success green, ✗ crimson, △ saffron, ◎ success green */
   142→function Cell({
   143→  type,
   144→  text,
   145→  jgateHighlight,
   146→}: {
   147→  type: CellType;
   148→  text: string;
   149→  jgateHighlight?: boolean;
   150→}) {
   151→  // If no type → plain text
   152→  if (!type) {
   153→    return (
   154→      <span
   155→        className={`font-inter text-[13px] ${jgateHighlight ? "font-semibold text-ink" : "text-slate"}`}
   156→      >
   157→        {text}
   158→      </span>
   159→    );
   160→  }
   161→  // Symbol + text
   162→  const symbol =
   163→    type === "check" ? (
   164→      <Check className="h-3.5 w-3.5 text-success" strokeWidth={2.5} />
   165→    ) : type === "cross" ? (
   166→      <span className="font-serif-jp text-[14px] font-bold text-mist leading-none">✗</span>
   167→    ) : type === "good" ? (
   168→      <span className="font-serif-jp text-[14px] font-bold text-success leading-none">◎</span>
   169→    ) : (
   170→      <span className="font-serif-jp text-[14px] font-bold text-saffron leading-none">△</span>
   171→    );
   172→  return (
   173→    <div className="flex items-start gap-2">
   174→      <span className="mt-0.5 shrink-0">{symbol}</span>
   175→      <span
   176→        className={`font-inter text-[13px] leading-relaxed ${
   177→          jgateHighlight ? "font-semibold text-ink" : "text-slate"
   178→        }`}
   179→      >
   180→        {text}
   181→      </span>
   182→    </div>
   183→  );
   184→}
