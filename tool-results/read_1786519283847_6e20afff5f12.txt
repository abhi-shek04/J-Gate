    37→/* 18 CSS particles drifting upward via jg-drift keyframe */
    38→function Particles() {
    39→  const particles = Array.from({ length: 18 }).map((_, i) => ({
    40→    left: `${(i * 53 + 7) % 100}%`,
    41→    size: 1.2 + ((i * 7) % 4) * 0.6,
    42→    delay: `${(i * 1.4) % 22}s`,
    43→    duration: `${18 + ((i * 5) % 12)}s`,
    44→  }));
    45→  return (
    46→    <div className="absolute inset-0 overflow-hidden" aria-hidden>
    47→      {particles.map((p, i) => (
    48→        <span
    49→          key={i}
    50→          className="absolute bottom-0 rounded-full bg-white"
    51→          style={{
    52→            left: p.left,
    53→            width: p.size,
    54→            height: p.size,
    55→            animation: `jg-drift ${p.duration} linear infinite`,
    56→            animationDelay: p.delay,
    57→          }}
    58→        />
    59→      ))}
    60→    </div>
    61→  );
    62→}

    64→/* Floating glass location badge (hero) */
    65→function GlassBadge({
    66→  emoji,
    67→  primary,
    68→  secondary,
    69→  accent,
    70→}: {
    71→  emoji: string;
    72→  primary: string;
    73→  secondary: string;
    74→  accent: "saffron" | "crimson" | "slate";
    75→}) {
    76→  const accentClass =
    77→    accent === "saffron"
    78→      ? "text-saffron border-saffron/30"
    79→      : accent === "crimson"
    80→        ? "text-crimson border-crimson/30"
    81→        : "text-mist border-white/15";
    82→  return (
    83→    <div className="glass-dark flex items-center gap-2.5 rounded-full px-4 py-2">
    84→      <span className="font-serif-jp text-[18px] leading-none">{emoji}</span>
    85→      <div className="flex flex-col leading-tight">
    86→        <span className={`font-inter text-[12px] font-bold uppercase ${accentClass.split(" ")[0]}`} style={{ letterSpacing: "0.08em" }}>
    87→          {primary}
    88→        </span>
    89→        <span className="font-inter text-[11px] text-mist">{secondary}</span>
    90→      </div>
    91→    </div>
    92→  );
    93→}
